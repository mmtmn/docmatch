#   path:<regex de caminho>    — arquivos ignorados por inteiro
#   literal:<regex>            — ocorrências permitidas em qualquer arquivo

path:tools/lint/

path:.*(package|package-lock|pnpm-lock|go|go\.sum|Cargo\.lock|Gemfile|Podfile)(\.json|\.yaml|\.lock|\.mod|\.sum)?$

path:\.github/
path:.*gradle.*

path:.*/LICENSE$
